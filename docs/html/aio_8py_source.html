76def run_build(args):
77 if not args:
78 print("Error: Missing build system (cython, cmake, ninja, meson, cargo).")
79 return
80
81 system = args[0].lower()
82 build_args = args[1:]
83
84 print(f"--- Build System: {system.upper()} ---")
85
86 if system == "cython":
87 if not HAS_CYTHON:
88 print("Error: cython module not available. (pip install cython)")
89 return
90 if not build_args:
91 print("Usage: aio.py build cython <file.py>")
92 return
93
94 target = build_args[0]
95 print(f"Cythonizing {target}...")
96 try:
97 # Create a dynamic setup script for cythonization
98 with tempfile.NamedTemporaryFile(mode='w', suffix='.py', delete=False) as f:
99 f.write(f"from setuptools import setup\nfrom Cython.Build import cythonize\nsetup(ext_modules=cythonize('{target}'))\n")
100 setup_file = f.name
101
102 subprocess.run([PYTHON_EXE, setup_file, "build_ext", "--inplace"], check=True)
103 os.remove(setup_file)
104 print("Cython build complete.")
105 except Exception as e:
106 print(f"Cython build failed: {e}")
107
108 elif system == "cmake":
109 if len(build_args) < 2:
110 print("Usage: aio.py build cmake <source_dir> <build_dir>")
111 return
112 subprocess.run(["cmake", "-S", build_args[0], "-B", build_args[1]])
113
114 elif system == "ninja":
115 if len(build_args) < 1:
116 print("Usage: aio.py build ninja <build_dir>")
117 return
118 subprocess.run(["ninja", "-C", build_args[0]])
119
120 elif system == "meson":
121 if len(build_args) < 2:
122 print("Usage: aio.py build meson <build_dir> <source_dir>")
123 return
124 subprocess.run(["meson", "setup", build_args[0], build_args[1]])
125
126 elif system == "cargo":
127 target = build_args[0] if build_args else "."
128 subprocess.run(["cargo", "build", "--manifest-path", os.path.join(target, "Cargo.toml")])
129
130 else:
131 print(f"Unknown build system: {system}")

268def ensure_package(package_name: str) -> bool:
269 """Check if a Python package is installed; if not, install it via pip."""
270 if importlib.util.find_spec(package_name) is None:
271 try:
272 subprocess.check_call([PYTHON_EXE, "-m", "pip", "install", package_name])
273 return True
274 except subprocess.CalledProcessError:
275 return False
276 return True

286 return [v.strip() for v in versions_str.split(",") if v.strip()]

335def _load_cache():
336 try:
337 with open(CACHE_FILE, "r") as f:
338 return json.load(f)

343 try:
344 with open(CACHE_FILE, "w") as f:

349def _cached_fetch(key, ttl, fetch_fn):
350 """Fetch via fetch_fn, caching the result on disk for `ttl` seconds."""
351 cache = _load_cache()
352 entry = cache.get(key) or {}
353 if time.time() - entry.get("ts", 0) < ttl and entry.get("items"):
354 return entry["items"]
355 try:
356 items = fetch_fn()
357 except Exception:
358 return entry.get("items", [])
359 if items:
360 cache[key] = {"ts": time.time(), "items": items}
361 _save_cache(cache)
362 return items

364def fetch_pypi_public_games():
365 """All pygame-related packages published on PyPI (via the Simple API)."""
366 def _scan():
367 import urllib.request
368 import re
369 url = "https://pypi.org/simple/"
370 req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"})
371 names = []
372 with urllib.request.urlopen(req, timeout=90) as resp:
373 buf = b""
374 while True:
375 chunk = resp.read(1024 * 256)
376 if not chunk:
377 break
378 buf += chunk
379 while b"</a>" in buf:
380 item, buf = buf.split(b"</a>", 1)
381 m = re.search(rb"<a[^>]*>([^<]+)", item)
382 if m:
383 name = m.group(1).decode("utf-8", "replace").strip()
384 if name and "pygame" in name.lower():
385 names.append(name)
386 return sorted(set(names))
387 return _cached_fetch("pypi_pygame", 7 * 24 * 3600, _scan)

389def fetch_github_pygame_repos():
390 """Public pygame game repositories on GitHub (top by stars, cached)."""
391 def _search():
392 import urllib.request
393 import json as _json
394 repos = {}
395 for query in ("pygame+language:python", "pygame+game"):
396 url = ("https://api.github.com/search/repositories"
397 f"?q={query}&sort=stars&order=desc&per_page=100")
398 req = urllib.request.Request(url, headers={
399 "User-Agent": "aiorewrite",
400 "Accept": "application/vnd.github+json",
401 })
402 with urllib.request.urlopen(req, timeout=30) as resp:
403 data = _json.loads(resp.read().decode("utf-8"))
404 for it in data.get("items", []):
405 repos[it["full_name"]] = {
406 "repo": it["full_name"],
407 "stars": it.get("stargazers_count", 0),
408 "desc": (it.get("description") or "").strip(),
409 }
410 return sorted(repos.values(), key=lambda r: -r["stars"])
411 return _cached_fetch("github_pygame", 24 * 3600, _search)

459 n += 1

461 for repo in games["github"]:

463 entries.append((n, "github", repo["repo"]))

469def resolve_games_entry(ref):
470 """Resolve a game name or listing number to a {kind, number, label} entry."""
471 if isinstance(ref, int) or (isinstance(ref, str) and ref.strip().isdigit()):
472 num = int(ref)
473 for number, kind, label in get_numbered_games():
474 if number == num:
475 return {"kind": kind, "number": number, "label": label}
476 return None
477 r = ref.strip().lower()
478 for name in get_installed_pygame_examples():
479 if name.lower() == r:
480 return {"kind": "builtin", "number": None, "label": name}
481 for name in fetch_pypi_public_games():
482 if name.lower() == r:
483 return {"kind": "pypi", "number": None, "label": name}
484 for repos in (fetch_github_pygame_repos(), fetch_awesome_pygame_repos()):
485 for repo in repos:

487 kind = "github" if repo["stars"] else "awesome"
488 return {"kind": kind, "number": None, "label": repo["repo"]}

876def parse_missing_requirements(output):
877 """Extract pip package names from a failed launch."""
878 needs = set()
879 for m in re.findall(r"The \w+ example requires: ([^\n]+)", output):
880 needs.update(m.split())
881 for m in re.findall(r"ModuleNotFoundError: No module named '([^']+)'", output):
882 mod = m.split(".")[0]
883 needs.add(MODULE_TO_PACKAGE.get(mod, mod))
884 for m in re.findall(r"ImportError: No module named ([^\s]+)", output):
885 mod = m.split(".")[0]
886 needs.add(MODULE_TO_PACKAGE.get(mod, mod))
887 return sorted(needs)

889def pip_install(packages):
890 """Install the given pip packages, returning success."""
891 pkgs = sorted({p.lower() for p in packages if p})
892 if not pkgs:
893 return False
894 print(f"Auto-installing missing requirements: {', '.join(pkgs)} ...")
895 try:
896 result = subprocess.run([PYTHON_EXE, "-m", "pip", "install", *pkgs],
897 text=True)
898 return result.returncode == 0
899 except Exception as e:
900 print(f"Auto-install failed: {e}")
901 return False

903def run_with_auto_install(cmd, label, cwd=None):
904 """Run a command; on failure auto-install missing requirements and retry once."""
905 try:
906 result = subprocess.run(cmd, capture_output=True, text=True, cwd=cwd)
907 except Exception as e:
908 print(f"Failed to launch {label}: {e}")
909 return
910
911 output = (result.stdout or "") + (result.stderr or "")
912 missing = parse_missing_requirements(output)
913 if result.returncode != 0 or missing:
914 if missing:
915 print(output.strip())
916 if pip_install(missing):
917 print(f"Retrying {label} ...")
918 result = subprocess.run(cmd, capture_output=True, text=True, cwd=cwd)
919 output = (result.stdout or "") + (result.stderr or "")

921 if result.stdout:
922 print(result.stdout, end="" if result.stdout.endswith("\n") else "\n")
923 if result.stderr:
924 print(result.stderr, end="" if result.stderr.endswith("\n") else "\n")

943 score += 1

952def find_entry_point(repo_dir):
953 candidates = []
954 for name in ENTRY_POINT_NAMES:
955 p = os.path.join(repo_dir, name)
956 if os.path.isfile(p):
957 candidates.append(p)
958 try:
959 for f in sorted(os.listdir(repo_dir)):
960 if f.endswith(".py") and os.path.isfile(os.path.join(repo_dir, f)):
961 if f not in ("setup.py", "setupx.py"):
962 candidates.append(os.path.join(repo_dir, f))
963 except OSError:
964 pass
965 for sub in ENTRY_POINT_DIRS:
966 d = os.path.join(repo_dir, sub)
967 if os.path.isdir(d):
968 for root, _dirs, files in os.walk(d):
969 depth = len(os.path.relpath(root, d).split(os.sep))
970 if depth > 3:
971 _dirs[:] = []
972 for f in sorted(files):
973 if f.endswith(".py") and os.path.isfile(os.path.join(root, f)):
974 candidates.append(os.path.join(root, f))
975 scored = [(_script_pygame_score(c), c) for c in candidates]
976 scored.sort(key=lambda t: (-t[0], len(t[1].split(os.sep)), len(os.path.basename(t[1])), t[1]))
977 if scored and scored[0][0] > 0:
978 return scored[0][1]
979 demo_dirs = ("examples", "example", "demo", "demos", "apps", "samples")
980 runnable_names = ("demo", "example", "main", "game", "play", "run")
981 for _score, c in scored:
982 parts = c.lower().split(os.sep)
983 if any(d in parts for d in demo_dirs) and \
984 os.path.basename(c).lower().startswith(runnable_names):
985 return c
986 return None

1039def find_module_name(dist_name):
1040 """Resolve the importable module name(s) for an installed distribution.
1041
1042 PyPI package names frequently differ from their module names (e.g. the
1043 'PyGameLab' distribution provides the 'pygamelab' module), so the
1044 launch logic resolves the real module before running ``python -m``.
1045 Returns a list ordered by likelihood; the first importable entry wins.
1046 """
1047 candidates = [dist_name]
1048 try:
1049 from importlib.metadata import packages_distributions
1050 for mod, dists in packages_distributions().items():
1051 if any(d.lower() == dist_name.lower() for d in dists) and mod not in candidates:
1052 candidates.append(mod)
1053 except Exception:
1054 pass
1055 try:
1056 from importlib.metadata import files
1057 for f in files(dist_name) or []:
1058 if str(f).endswith("top_level.txt"):
1059 for ln in (f.read_text() or "").splitlines():
1060 ln = ln.strip()
1061 if ln and ln not in candidates:
1062 candidates.append(ln)
1063 except Exception:
1064 pass
1065 ordered = sorted(candidates, key=lambda c: (c.lower() != dist_name.lower(), c.lower()))
1066 importable = [c for c in ordered if importlib.util.find_spec(c) is not None]
1067 return importable or ordered

1069def _is_runnable_module(mod):
1070 """True if `python -m <mod>` can execute it (module, or package with __main__)."""
1071 try:
1072 spec = importlib.util.find_spec(mod)
1073 except Exception:
1074 return False
1075 if spec is None:
1076 return False
1077 if spec.submodule_search_locations is not None:
1078 try:
1079 return importlib.util.find_spec(mod + ".__main__") is not None
1080 except Exception:
1081 return False
1082 return True
